# LangGraph Memory Filter Dependencies
# Install with: pip install -r requirements-langgraph.txt

# Core LangGraph dependencies
# IMPORTANT: stay on the langgraph 0.6.x line (verified 2026-10-13 against langchain-core 1.4.8).
# langgraph 1.x pulls langgraph-sdk 0.4.x, which pins websockets<17; the Open WebUI 0.12.0 image
# ships websockets 17.2, so the startup pip install downgraded websockets under the running
# server and the first import of this filter failed with "cannot import name 'OP_BINARY'".
# langgraph-sdk 0.2/0.3 has no websockets dependency at all.
# (langgraph 0.2.45 is still incompatible with modern langchain-core: RunnableSerializable error.)
langgraph~=0.6.11
langgraph-sdk<0.4
langgraph-checkpoint-postgres<3

# PostgreSQL driver with binary backend
# psycopg[binary] includes the pre-compiled binary backend driver
# This avoids needing PostgreSQL development libraries for compilation
psycopg[binary]>=3.1.0

# Connection pooling - required by the filter
psycopg-pool>=3.1.0

# Quick install command:
# pip install "langgraph~=0.6.11" "langgraph-sdk<0.4" "langgraph-checkpoint-postgres<3" "psycopg[binary]" psycopg-pool
